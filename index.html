Revisa la consola (F12). `; } } // ========================================== // PRE-REGISTRO CON QR // ========================================== async function guardarRegistroQR() { const qrInput = document.getElementById("qr-code-input"); const nameInput = document.getElementById("qr-name-input"); const qrCode = qrInput.value.trim(); const nombre = nameInput.value.trim(); if (!qrCode || !nombre) { alert("⚠️ Por favor complete tanto el código QR como el Nombre."); return; } try { const { data, error } = await supabaseClient .from("usuarios") .upsert([{ id: qrCode, nombre: nombre }], { onConflict: "id" }); if (error) { alert("❌ Error al registrar: " + error.message); } else { alert("✅ ¡Registro con QR exitoso! Verifique el código de la pantalla."); } else { alert(`✅ ¡Bienvenido ${nombre}! Tu carnet quedó vinculado con éxito.`); pinInput.value = ""; nameInput.value = ""; closeModal("modal-pin"); cargarLeaderboard(); // Recargar tabla } } catch (err) { alert("❌ Error inesperado: " + err.message); } } // ========================================== // INICIALIZACIÓN Y EVENTOS // ========================================== document.addEventListener("DOMContentLoaded", () => { // Cargar datos por primera vez cargarLeaderboard(); // Auto-refrescar cada 4 segundos (Ideal para mostrar cambios en vivo durante la feria) setInterval(cargarLeaderboard, 4000); // Asignar eventos a los botones de los modales const btnSaveQR = document.getElementById("btn-save-qr"); if (btnSaveQR) btnSaveQR.addEventListener("click", guardarRegistroQR); const btnSavePIN = document.getElementById("btn-save-pin"); if (btnSavePIN) btnSavePIN.addEventListener("click", vincularConPIN); });
